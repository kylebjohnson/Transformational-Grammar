% ===========================================================

\newcommand{\class}{f}			% Class
\newcommand{\sdate}{f}			% Date
% ===========================================================

% ======================================================================
% = Sets up a header that puts Johnson, class, lecture and date across =
% ======================================================================
\pagestyle{fancy}
\rhead[\textsf{\sdate}]{\textsf{\sdate}}
\lhead[\textsf{Johnson: \class}]{\textsf{Johnson: \class}}
\chead[\textsf{\lecture}]{\textsf{\lecture}}

% ======================================================================

% ======================================================
% = Produces a footer with a centered page number =
% ======================================================
\cfoot[\textsf{\large \thepage}]{\textsf{\large \thepage}}
% ======================================================================


\begin{document}
	
\thispagestyle{empty}

% ================================
% = This sets up the first page  =
% ================================

    {\sffamily\Large \lecture %lecture
    }%
  \end{flushleft} \vspace*{-.17in}
  \rule{\linewidth}{.5mm} \vspace*{-.25in}
  \begin{flushright}
    {\sffamily
      Kyle Johnson \hspace*{\fill} \class \\
    \sdate}%
    \end{flushright}
